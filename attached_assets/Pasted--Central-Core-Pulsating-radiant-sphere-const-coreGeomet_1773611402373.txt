

    // Central Core: Pulsating radiant sphere
    const coreGeometry = new THREE.SphereGeometry(2, 64, 64);
    const coreMaterial = new THREE.ShaderMaterial({
      uniforms: {
        time: { value: 0 },
      },
      vertexShader: `
        uniform float time;
        varying vec3 vNormal;
        varying vec3 vPosition;
        void main() {
          vNormal = normal;
          vPosition = position + normal * sin(time + length(position)) * 0.2;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(vPosition, 1.0);
        }
      `,
      fragmentShader: `
        uniform float time;
        varying vec3 vPosition;
        void main() {
          float pulse = sin(length(vPosition) * 5.0 + time) * 0.5 + 0.5;
          vec3 color = mix(vec3(0.9, 0.2, 1.0), vec3(0.2, 0.6, 1.0), pulse);
          gl_FragColor = vec4(color, 1.0);
        }
      `,
    });
    const core = new THREE.Mesh(coreGeometry, coreMaterial);
    scene.add(core);

    // Concentric rings: Flowing luminous energy waves
    const ringsGroup = new THREE.Group();
    for (let i = 1; i <= 6; i++) {
      const ringGeometry = new THREE.TorusGeometry(i * 1.5, 0.08, 16, 100);
      const ringMaterial = new THREE.MeshStandardMaterial({
        color: new THREE.Color(`hsl(${i * 60}, 70%, 50%)`),
        emissive: new THREE.Color(`hsl(${i * 60}, 90%, 80%)`),
        transparent: true,
        opacity: 0.5,
      });
      const ring = new THREE.Mesh(ringGeometry, ringMaterial);
      ring.rotation.x = Math.PI / 2;
      ringsGroup.add(ring);
      gsap.to(ring.rotation, {
        z: Math.PI * 2,
        duration: 10 - i,
        repeat: -1,
        ease: "linear",
      });
    }
    scene.add(ringsGroup);

    // Plasma threads: Tentacle-like strands of pure energy
    const threadGroup = new THREE.Group();
    for (let i = 0; i < 40; i++) {
      const threadGeometry = new THREE.CatmullRomCurve3([
        new THREE.Vector3(0, 0, 0),
        new THREE.Vector3(Math.random() * 4 - 2, Math.random() * 4 - 2, Math.random() * 4 - 2),
        new THREE.Vector3(Math.random() * 8 - 4, Math.random() * 8 - 4, Math.random() * 8 - 4),
      ]).getPoints(50);
      const thread = new THREE.Line(
        new THREE.BufferGeometry().setFromPoints(threadGeometry),
        new THREE.LineBasicMaterial({
          color: new THREE.Color(`hsl(${Math.random() * 360}, 80%, 60%)`),
          linewidth: 4,
        })
      );
      threadGroup.add(thread);
      gsap.to(thread.position, {
        x: Math.random() * 2,
        y: Math.random() * 2,
        z: Math.random() * 2,
        duration: 5,
        repeat: -1,
        yoyo: true,
        ease: "sine.inOut",
      });
    }
    scene.add(threadGroup);
