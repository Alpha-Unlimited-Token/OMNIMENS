Below is one complete example that builds on the snippet you mentioned. It uses the pyaudio library to record ambient sound from your microphone, saves the recording as a WAV file, and then utilizes the SpeechRecognition library (with Google’s web API) to transcribe the audio into text. You can adjust parameters such as recording duration, sample rate, or channels to suit your needs.

#!/usr/bin/env python3
import pyaudio
import wave
import speech_recognition as sr

def record_audio(duration=5, filename="ambient.wav"):
    # Recording parameters:
    chunk = 1024 # size of each audio chunk
    sample_format = pyaudio.paInt16 # 16-bit samples
    channels = 1 # mono audio
    fs = 44100 # sampling rate (Hz)

    p = pyaudio.PyAudio()
    stream = p.open(format=sample_format,
                    channels=channels,
                    rate=fs,
                    frames_per_buffer=chunk,
                    input=True)

    print("Recording ambient sound...")
    frames = []
    for i in range(0, int(fs / chunk * duration)):
        data = stream.read(chunk)
        frames.append(data)
    print("Finished recording.")

    # Stop and close the stream
    stream.stop_stream()
    stream.close()
    p.terminate()

    # Save the recorded data as a WAV file
    wf = wave.open(filename, 'wb')
    wf.setnchannels(channels)
    wf.setsampwidth(p.get_sample_size(sample_format))
    wf.setframerate(fs)
    wf.writeframes(b''.join(frames))
    wf.close()

def speech_to_text(filename="ambient.wav"):
    recognizer = sr.Recognizer()
    
    # Load the WAV file
    with sr.AudioFile(filename) as source:
        audio = recognizer.record(source)
    
    try:
        # Using Google's speech recognition service
        text = recognizer.recognize_google(audio)
        print("Transcription:")
        print(text)
    except sr.UnknownValueError:
        print("Speech Recognition could not understand the audio.")
    except sr.RequestError as e:
        print(f"Could not request results from the speech recognition service; {e}")

if __name__ == "__main__":
    record_audio(duration=5, filename="ambient.wav")
    speech_to_text(filename="ambient.wav")

This code works as follows:
1. The record_audio() function captures a few seconds of audio input from your microphone and writes it into the file "ambient.wav".
2. The speech_to_text() function then loads that file and attempts to transcribe the audio using Google’s speech recognition API.
3. Finally, running the script records the ambient sound and prints out the transcription.
